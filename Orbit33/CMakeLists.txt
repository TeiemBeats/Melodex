cmake_minimum_required(VERSION 3.22)
project(Orbit33 VERSION 0.1.0)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(Orbit33
 COMPANY_NAME "Mark33s Beats Labs" BUNDLE_ID "com.mark33sbeatslabs.orbit33"
 PLUGIN_MANUFACTURER_CODE M33B PLUGIN_CODE Orb3
 FORMATS VST3 Standalone PRODUCT_NAME "Orbit33"
 IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(Orbit33 PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(Orbit33 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Orbit33 PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_audio_processors juce::juce_gui_extra)
juce_generate_juce_header(Orbit33)
include(CTest)
if(BUILD_TESTING)
 add_executable(orbit33_presets tests/presets.cpp)
 if(MSVC)
  target_compile_options(orbit33_presets PRIVATE /UNDEBUG)
 else()
  target_compile_options(orbit33_presets PRIVATE -UNDEBUG)
 endif()
 add_test(NAME factory-presets COMMAND orbit33_presets)
 juce_add_console_app(orbit33_tests PRODUCT_NAME "Orbit33 checks")
 target_sources(orbit33_tests PRIVATE tests/plugin.cpp)
 target_include_directories(orbit33_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/Orbit33_artefacts/JuceLibraryCode")
 target_link_libraries(orbit33_tests PRIVATE Orbit33 juce::juce_audio_utils juce::juce_dsp juce::juce_gui_extra)
 add_test(NAME instrument-midi-state COMMAND orbit33_tests)
endif()
