cmake_minimum_required(VERSION 3.22)
project(MELODEX VERSION 0.2.0)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(MELODEX COMPANY_NAME "Mark33s Beats Labs" BUNDLE_ID "com.markeesbeatslabs.melodex" PLUGIN_MANUFACTURER_CODE MkBL PLUGIN_CODE MldX FORMATS VST3 PRODUCT_NAME "MELODEX" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(MELODEX PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(MELODEX PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(MELODEX PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_audio_processors juce::juce_gui_extra)
juce_generate_juce_header(MELODEX)


include(CTest)
if(BUILD_TESTING)
 add_executable(preset_tests tests/presets.cpp)
 if(MSVC)
  target_compile_options(preset_tests PRIVATE /UNDEBUG)
 else()
  target_compile_options(preset_tests PRIVATE -UNDEBUG)
 endif()
 add_test(NAME factory-presets COMMAND preset_tests)
endif()
if(BUILD_TESTING)
 juce_add_console_app(plugin_tests PRODUCT_NAME "Melodex checks")
 target_sources(plugin_tests PRIVATE tests/plugin.cpp)
 target_include_directories(plugin_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/MELODEX_artefacts/JuceLibraryCode")
 target_link_libraries(plugin_tests PRIVATE MELODEX juce::juce_audio_utils juce::juce_dsp juce::juce_gui_extra)
 add_test(NAME plugin-midi-state COMMAND plugin_tests)
endif()
